                function checkPropforData(geoRow ,checkData ) {
                    var matchedIndex ;

                    var geoProps = ["adm_cd" , "adm_nm" , "alias1", "alias2", "alias3"] // Object.keys (geoRow)
                    for (var k = 0 ; k < checkData.length ; k ++ ) {
                        for (var l = 0 ; l < geoProps.length ; l ++ ) {
                            if (geoRow[geoProps[l]]==undefined ) {break ; }
                            if (checkData[k].name == geoRow[geoProps[l]]) {
                                matchedIndex = k ;
                                break ;
                            }
                            else if (checkData[k].name.split(" ").reverse()[0] == geoRow[geoProps[l]]){
                                matchedIndex = k ;
                                break ;
                            }
                            else if (checkData[k].name.replaceAll (" ", "") == geoRow[geoProps[l]].replaceAll(" ","")){
                                matchedIndex = k ;
                                break ;
                            }
                        }
                    }
                    return matchedIndex
                }

                function makegeojsonArray(loadedJson ) {
                    var returnArray = [] ;

                        if ( dispObject.currentMode == $MAPMODE.SIDO ) {

                            alias = sidoAlias [i.properties.adm_cd] && sidoAlias [i.properties.adm_cd][custProps.aliasSido] ;
                        }
                        else
                        {alias = i.properties.adm_nm.split(" ").reverse()[0] ;}
                        returnArray.push({
                            "adm_cd": i.properties.adm_cd,
                            "adm_nm": i.properties.adm_nm,
                            "alias1" : alias,
                            "x" : i.properties.x,
                            "y" : i.properties.y,
                            properties: i.properties
                        });
                    }
                    return returnArray ;
                }

            function getColorStyle(color) {
                return { color : color.fillColor ,
                    opacity : color.fillColor == 'transparent' ? 0 : color.fillAlpha /100 };
            }

            function getFontStyle (fontStyle) {
                var cssFontStyle = {} ;
                cssFontStyle.color = fontStyle.fontColor ;
                cssFontStyle.fontWeight = fontStyle.fontWeight ? "bold" : "normal" ;
                cssFontStyle.fontStyle = fontStyle.fontItalic ? "italic" : "normal";
                cssFontStyle.fontFamily = fontStyle.fontFamily;
                cssFontStyle.fontSize = parseInt(fontStyle.fontSize) ;
                cssFontStyle["text-decoration-line"] = fontStyle.fontUnderline == "true" ? "underline" : "" ;
                cssFontStyle["text-decoration-line"] = cssFontStyle["text-decoration-line"] + " "
                    + fontStyle.fontLineThrough == "true" ? "line-through" : "" ;
                return cssFontStyle ;
            }

            function getBorderStyle (propStyle) {
                var borderStyle = {} ;
                switch (parseInt(propStyle.lineStyle)) {
                    case $LINETYPE.NONE :
                        borderStyle = {width : 0 ,type : 'solid' }; break ;
                    case $LINETYPE.THIN :
                        borderStyle = {width : 1 ,type : 'solid' }; break ;
                    case $LINETYPE.DASHED :
                        borderStyle = {width : 1 ,type : 'dashed' }; break ;
                    case $LINETYPE.DOTTED :
                        borderStyle = {width : 1 ,type : 'dotted' }; break ;
                    case $LINETYPE.THICK :
                        borderStyle = {width : 2 ,dasharray : 0 }; break ;
                }
                borderStyle.lineColor = propStyle.lineColor ;
                return borderStyle ;
            }

            function hexToRGB (hex, alpha) {
                if (!hex) {
                    return "transparent" ;
                }
                if (hex.toLowerCase() == "transparent") {
                    return "transparent" ;
                }
                var r = parseInt(hex.slice(1, 3), 16);
                var g = parseInt(hex.slice(3, 5), 16);
                var b = parseInt(hex.slice(5, 7), 16);
                if (alpha >= 0) {
                    return "rgba( "+r+","+g+","+b+"," + alpha +")";
                } else {
                    return "rgba( "+r+","+g+","+b+" )";
                }
            }
